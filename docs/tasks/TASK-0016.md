# TASK-0016 · Transaction per request and short security writes

**Chain:** DATABASE (step 6 of 16) · **Release:** 1 · **Module:** Core · **Part:** Backend (D-187; a shared backend card with no door, AGENTS.md section 7) · **Depends on:** TASK-0015
**Requirement IDs covered:** NFR-021 (every save is all or nothing), CORE-014, TECH-DESIGN section 2 (step 5 and the security writes) and 6.6
**Builder:** Antigravity (while Codex has no tokens, D-194) · **Reviewer:** Claude

**Goal:** the sixth card of step 2.2. Every request that is not public runs in **one database transaction** with its company set, so everything it saves is kept together or not at all (NFR-021). **Two connection pools**, both signed in as the app account: the request pool (`DB_POOL_SIZE`, default 10) holds each request's transaction, and a small security pool (`DB_SECURITY_POOL_SIZE`, default 3) holds the **short security writes** (session end, last-activity time, per-person counter, two-step attempts, a hit limit's log entry), which commit even when the request is refused. A request never waits for a second connection from its own pool, so the API cannot lock itself when many requests arrive at once (CR-012). Both pools wait at most 10 seconds for a free connection (a full pool answers 500 instead of hanging), and the security pool's connections wait at most 5 seconds for a lock. The request's transaction is kept on the request context while it runs. **The shared hooks** (`createHooks()`): the transaction step is real; every step a later card builds refuses each request until then (500), and the feature switch lets every feature through until TASK-0057. The app mounts its doors with these hooks.

1. Create `apps/api/src/db/database.ts` with exactly this content:
   ```ts
   import { Pool, type PoolConfig } from 'pg';
   import { drizzle } from 'drizzle-orm/node-postgres';
   import { getContext } from '../shared/context.js';
   import { withCompany, type Database, type Transaction } from './company.js';

   // The app's two connection pools, both signed in as nexus_app (TECH-DESIGN section 2, steps 5 to 8). The request
   // pool holds each request's transaction; the small security pool holds the short security writes, which commit
   // whatever happens to the request. A request never waits for a second connection from its own pool, so the API
   // cannot lock itself when many requests arrive at once (CR-012). Nothing here reads a setting or opens a
   // connection until first use.

   type Settings = Record<string, string | undefined>;

   // An optional pool size (TECH-DESIGN 6.6): a whole number above 0, or the default.
   export function poolSize(value: string | undefined, fallback: number): number {
     const size = Number(value);
     return Number.isInteger(size) && size > 0 ? size : fallback;
   }

   // Both pools wait at most 10 seconds for a free connection, so a full pool answers 500 instead of hanging; the
   // security pool's connections wait at most 5 seconds for a lock.
   export function poolSettings(settings: Settings = process.env): {
     request: PoolConfig;
     security: PoolConfig;
   } {
     const value = settings.DATABASE_URL_APP;
     if (!value) throw new Error('DATABASE_URL_APP is not set');
     if (!URL.canParse(value)) throw new Error('DATABASE_URL_APP is not a database address');
     const app = new URL(value);
     if (app.username !== 'nexus_app') throw new Error('DATABASE_URL_APP must sign in as nexus_app');
     const address: PoolConfig = {
       host: app.hostname,
       port: Number(app.port) || 5432,
       user: app.username,
       password: decodeURIComponent(app.password),
       database: app.pathname.slice(1),
       connectionTimeoutMillis: 10_000,
     };
     return {
       request: { ...address, max: poolSize(settings.DB_POOL_SIZE, 10) },
       security: { ...address, max: poolSize(settings.DB_SECURITY_POOL_SIZE, 3), lock_timeout: 5_000 },
     };
   }

   interface Pools {
     request: Pool;
     security: Pool;
     db: Database;
     securityDb: Database;
   }

   let pools: Pools | undefined;

   function getPools(): Pools {
     if (!pools) {
       const settings = poolSettings();
       const request = new Pool(settings.request);
       const security = new Pool(settings.security);
       // A connection that breaks while idle is dropped from its pool and replaced on next use; without a listener
       // the error would stop the API.
       request.on('error', () => {});
       security.on('error', () => {});
       pools = { request, security, db: drizzle(request), securityDb: drizzle(security) };
     }
     return pools;
   }

   // Step 5: runs the rest of a request in one transaction on the request pool, with the request's company set
   // (TECH-DESIGN 5.1). The transaction is on the request context while it runs, and the repositories read it there.
   export async function inRequestTransaction<T>(run: () => Promise<T>): Promise<T> {
     const context = getContext();
     if (!context.companyId) throw new Error('the request transaction needs the company');
     return withCompany(getPools().db, context.companyId, async (tx) => {
       context.tx = tx;
       try {
         return await run();
       } finally {
         delete context.tx;
       }
     });
   }

   // A short security write (TECH-DESIGN section 2): its own transaction on the security pool, with the company set,
   // committed whatever happens to the request. Inside it, use only its own transaction, never the request's, and
   // await nothing that needs the request pool.
   export async function securityWrite<T>(
     companyId: string,
     work: (tx: Transaction) => Promise<T>,
   ): Promise<T> {
     return withCompany(getPools().securityDb, companyId, work);
   }

   // Ends both pools (when the API stops, and in the tests); the next use makes them again from the settings.
   export async function closePools(): Promise<void> {
     const open = pools;
     pools = undefined;
     if (open) await Promise.all([open.request.end(), open.security.end()]);
   }
   ```
2. Create `apps/api/src/shared/hooks.ts` with exactly this content:
   ```ts
   import type { RouteHooks } from './route.js';
   import { inRequestTransaction } from '../db/database.js';

   // The shared steps of every non-public door (TECH-DESIGN section 2), which route.ts runs in order. Each card that
   // builds a step replaces its line here: TASK-0033 session, TASK-0034 idleTimeout, TASK-0037 twoStepGate,
   // TASK-0035 personLimit, TASK-0057 featureSwitch, TASK-0031 permission and hideFields, TASK-0022 idempotency.
   // A step not built yet refuses every request (500), so no door works without it. Making the hooks reads no
   // setting and opens no connection.
   function notBuilt(step: string): never {
     throw new Error(`${step} is not built yet`);
   }

   export function createHooks(): RouteHooks {
     return {
       session: async () => notBuilt('session'),
       inTransaction: async (_info, run) => inRequestTransaction(run),
       idleTimeout: async () => notBuilt('idleTimeout'),
       twoStepGate: async () => notBuilt('twoStepGate'),
       personLimit: async () => notBuilt('personLimit'),
       // On purpose, until TASK-0057 builds the feature switches: every feature counts as switched on. The
       // permission check after it still runs.
       featureSwitch: async () => {},
       permission: async () => notBuilt('permission'),
       idempotency: async () => notBuilt('idempotency'),
       hideFields: () => notBuilt('hideFields'),

   }
   ```
