4. Create `packages/contracts/src/contract.ts`:
   ```ts
   import type { ZodObject, ZodRawShape, ZodType } from 'zod';

   export type HttpMethod = 'get' | 'post' | 'put' | 'patch' | 'delete';

   /** One API door, written once and shared by the backend and the frontend (D-187). */
   export interface Contract {
     method: HttpMethod;
     /** OpenAPI style, for example '/api/core/users/{userId}'. */
     path: string;
     operationId: string;
     /** The Nexus module that owns the door: 'core', 'crm', … */
     module: string;
     summary: string;
     params?: ZodObject<ZodRawShape>;
     query?: ZodObject<ZodRawShape>;
     body?: ZodType;
     response: ZodType;
     /** Fields of the response that a role may hide (TECH-DESIGN 6.2). */
     hideableFields?: readonly string[];
   }

   export function defineContract<const C extends Contract>(contract: C): C {
     return contract;
   }
   ```
5. Create `packages/contracts/src/core/health.ts`:
   ```ts
   import { z } from 'zod';
   import { defineContract } from '../contract.js';

   export const HealthResponse = z.object({ ok: z.literal(true), service: z.literal('api') });

   export const getHealth = defineContract({
     method: 'get',
     path: '/api/health',
     operationId: 'getHealth',
     module: 'core',
     summary: 'Is the API running?',
     response: HealthResponse,
   });
   ```
6. Create `packages/contracts/src/all.ts`: `export const allContracts = [getHealth] as const;` (importing `getHealth` from `./core/health.js`). Every later card that adds a door adds it to this list.
7. Create `packages/contracts/src/openapi.ts` with exactly this code (the list is widened to `readonly Contract[]` here only, so later cards keep each door's own exact type):
   ```ts
   import { OpenAPIRegistry, OpenApiGeneratorV31 } from '@asteasolutions/zod-to-openapi';
   import { allContracts } from './all.js';
   import type { Contract } from './contract.js';

   export function generateOpenApi() {
     const registry = new OpenAPIRegistry();
     // Widen the const list so the optional params, query and body can be read.
     const contracts: readonly Contract[] = allContracts;
     for (const contract of contracts) {
       registry.registerPath({
         method: contract.method,
         path: contract.path,
         operationId: contract.operationId,
         summary: contract.summary,
         tags: [contract.module],
         request: {
           ...(contract.params ? { params: contract.params } : {}),
           ...(contract.query ? { query: contract.query } : {}),
           ...(contract.body ? { body: { content: { 'application/json': { schema: contract.body } } } } : {}),
         },
         responses: { 200: { description: 'OK', content: { 'application/json': { schema: contract.response } } } },
       });
     }
     return new OpenApiGeneratorV31(registry.definitions).generateDocument({
       openapi: '3.1.0',
       info: { title: 'YTP Nexus API', version: '0.1.0' },
     });
   }
   ```
8. Create `packages/contracts/src/index.ts`: export everything from `./contract.js`, `./core/health.js`, `./all.js` and `./openapi.js`.
9. Create `packages/contracts/scripts/write-openapi.ts`: writes `JSON.stringify(generateOpenApi(), null, 2)` plus a final newline to `packages/contracts/openapi.json` (the path is worked out from the script's own folder, not the current directory) and prints `openapi.json written`.
