   import { useSyncExternalStore } from 'react';
   import { authClient } from './auth-client';

   // Who is signed in, until TASK-0046 and TASK-0052 (TASK-0049): exactly one GET /api/auth/get-session per page load,
   // made outside React the first time a screen asks, so StrictMode's second run never calls it again. Its answer is
   // never kept: only whether someone is signed in. A good sign-in says signed in; a 401 from a Nexus door says signed out.
   export type SessionState = 'checking' | 'signed-in' | 'signed-out';

   let state: SessionState = 'checking';
   let check: Promise<void> | undefined;
   const listeners = new Set<() => void>();

   function setState(next: SessionState): void {
     state = next;
     for (const listener of listeners) listener();
   }

   function checkSession(): void {

       ({ data }) => {

     };
   }
