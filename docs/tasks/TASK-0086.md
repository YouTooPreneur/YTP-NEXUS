   ```ts
   import type { RequestHandler } from 'express';
   import { pino } from 'pino';
   import { pinoHttp } from 'pino-http';
   import { getContext } from './context.js';

   const SECRET_KEYS = ['password', 'code', 'token', 'secret', 'newPassword', 'currentPassword'];

   // Removed from every log line: sign-in headers, the Referer and redirect addresses (they can carry a
   // one-time token in their query), and secret fields at the top level and up to three levels deep.
   export const REDACT_PATHS = [
     'req.headers.cookie',
     'req.headers.authorization',
     'req.headers.referer',
     'res.headers["set-cookie"]',
     'res.headers.location',
     ...SECRET_KEYS,
     ...SECRET_KEYS.map((key) => `*.${key}`),
     ...SECRET_KEYS.map((key) => `*.*.${key}`),
   ];

   type ErrorLike = {
     message: string;
     stack?: unknown;
     code?: unknown;
     cause?: unknown;
     raw?: unknown;
   };

   function isErrorLike(value: unknown): value is ErrorLike {
     return (
       typeof value === 'object' &&
       value !== null &&
       typeof (value as { message?: unknown }).message === 'string'
     );
   }

   function describeError(error: ErrorLike): { type: string; message: string; errorCode?: string } {
     const type = error.constructor?.name || 'Error';
     // A failed database query repeats every value it was given after "params:" in its message.
     const cut = error.message.indexOf('\nparams: ');
     const message = cut === -1 ? error.message : error.message.slice(0, cut);
     const code = error.code;
     return typeof code === 'string' || typeof code === 'number'
       ? { type, message, errorCode: String(code) }
       : { type, message };
   }

   // Logs an error's type, message (without a query's values), code, stack frames and one level of
   // cause. Every other property (params, input, detail, body, config) is dropped: it can hold secrets.
   export function cleanError(value: unknown): unknown {
     const error = isErrorLike(value) && value.raw instanceof Error ? value.raw : value;
     if (!isErrorLike(error)) return { type: typeof error };
     const described = describeError(error);
     const frames = String(error.stack ?? '')
       .split('\n')
       .filter((line) => /^\s+at /.test(line));
     return {
       ...described,
       stack: [`${described.type}: ${described.message}`, ...frames].join('\n'),
       ...(isErrorLike(error.cause) ? { cause: describeError(error.cause) } : {}),
     };
   }

   export function createLogger(destination?: pino.DestinationStream): pino.Logger {
     return pino(
       {
         level: process.env.LOG_LEVEL || 'info',
         redact: { paths: REDACT_PATHS, censor: '[removed]' },
         serializers: { err: cleanError },
         formatters: {
           // An error logged under any other top-level key ({ error }, { cause }) is cleaned the same way.
           log: (object) =>
             Object.fromEntries(
               Object.entries(object).map(([key, value]) => [
                 key,
                 key !== 'err' && value instanceof Error ? cleanError(value) : value,
               ]),
             ),
         },
       },
       destination,
     );
   }

   export function httpLogger(logger: pino.Logger): RequestHandler {
     return pinoHttp({
       logger,
       genReqId: () => getContext().requestId,
       serializers: {
         // pino-http hands these its own copy of the request, so the real request is not changed.
         req: (req: { url?: string; query?: unknown }) => {
           req.url = req.url?.split('?')[0];
           delete req.query;
           return req;
         },
         err: cleanError,
       },
     });
   }
   ```
   Why each part is there:
   - `REDACT_PATHS`: the old list removed secret fields only one level down; a reset or invite page sends its whole address, `?token=` included, in the Referer of its requests, and a redirect carries its address in `location`.
   - `cleanError`: pino's standard error printer writes every property of an error and its message as they are. A failed database query (Drizzle's `DrizzleQueryError`) repeats every value after `params:` in its message and in `params`; Node's address error keeps the whole address, password included, in `input`; a database error's `detail` names the person's email. The cleaner keeps only the type, the message without the values, the code (as `errorCode`, so the database's error number stays readable: `*.code` hides a field named `code`), the stack frames and one level of cause. pino-http gives its own logger its own error printer, so `httpLogger` sets it too, and pino-http hands the printer an already printed copy with the real error in `raw`.
   - `formatters.log`: pino runs `cleanError` only for the key `err`; an error logged under any other top-level key (`{ error }`, `{ cause }`) was written with every property it carries (`params`, `input`). This cleans every error at the top level of a log line the same way; `err` is left to its own printer, so pino-http's copy still reaches `cleanError` with the real error in `raw`.
   - The `req` printer: pino-http writes the request's address with its query and the query again as `query`; only the path is kept.
   - `process.env.LOG_LEVEL || 'info'`: with `??`, a blank `LOG_LEVEL` (a line `LOG_LEVEL=` in a settings file) stopped the API at start-up.
