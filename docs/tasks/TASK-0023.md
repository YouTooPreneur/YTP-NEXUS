**Requirement IDs covered:** NFR-020 (rate limits: one address's flood is slowed while others are not, refused requests count, a hit limit is logged), D-124, DATA-MODEL 3.4 (`core.rate_counters`) and 3.3 (the `limit_hit` event), TECH-DESIGN section 2 step 2 (the per-address limit)
**Builder:** Codex (D-201) · **Reviewer:** Claude (planning session)

**Goal:** the fourteenth card of step 2.2. Before anyone signs in, Nexus cannot tell people apart, so the **sign-in doors** (`/api/auth/...`) and the **two-step doors** (`/api/core/two-step/...`) get a **limit per address**: every request from one address is counted in a fixed time window (by default 20 a minute, from `ADDRESS_LIMITS` in the environment file), **refused requests included**, and past the limit the door answers **429** "Too many tries. Please try again in N seconds." with a `Retry-After` header until the window ends. Other addresses are never slowed. The counts live in a new system table, `core.rate_counters` (one row per bucket, such as `signin:ip:203.0.113.7`, and window), and are added by one statement in a short write of its own on the security pool, using the database's clock, so a hundred requests at once are each counted exactly once. The first refused request of a window writes **one** `limit_hit` entry in the security log's system chain (with the address and browser), never one per refused request. The address counted is the one the proxy saw (Express trusts exactly one proxy), so a made-up first address in `X-Forwarded-For` changes nothing. The limiter sits right after the same-site check and before the body is read, so a cross-site request is refused before it is counted and a limited address's body is never read. No sign-in door exists yet (TASK-0032 builds them); the counter `hitLimit` is ready for the per-person limit (TASK-0035).

3. Create `apps/api/src/modules/core/repository/rate-counters.ts` with exactly this content:
   ```ts
import { sql } from 'drizzle-orm';
import { securityWriteWithoutCompany } from '../../../db/database.js';
import { writeSecurity } from './security-log.js';

// The counter of the limits (TASK-0023, NFR-020). hitLimit counts one request in its bucket's current window, refused
// requests included, in a short security write of its own with no company (never a lock in the request's transaction),
// and says whether the limit is passed. The window is fixed and worked out by the database's clock. The first request
// over the limit in a window writes one limit_hit entry in the security log's system chain, in the same write. It runs
// inside a request (the security log takes the address and browser from the request context).
export interface Limit {
  // The limit's name, for example signin.
  name: string;
  // Who is counted, for example ip:203.0.113.7.
  key: string;
  // The requests allowed in one window.
  max: number;
  windowSeconds: number;
}

export interface Counted {
  count: number;
  refused: boolean;
  // The seconds until the window ends, at least 1.
  retryAfterSeconds: number;
}

export async function hitLimit(limit: Limit): Promise<Counted> {
  const bucket = `${limit.name}:${limit.key}`;
  const seconds = limit.windowSeconds;
  return securityWriteWithoutCompany(async (tx) => {
    const result = await tx.execute(
      sql`INSERT INTO core.rate_counters AS counter (bucket, window_start, count)
          VALUES (${bucket}, to_timestamp(floor(extract(epoch FROM clock_timestamp()) / ${seconds}::int) * ${seconds}::int), 1)
          ON CONFLICT (bucket, window_start) DO UPDATE SET count = counter.count + 1
          RETURNING counter.count,
            ceil(extract(epoch FROM counter.window_start) + ${seconds}::int - extract(epoch FROM clock_timestamp()))::int AS retry_after`,
    );
    const row = result.rows[0];
    if (row === undefined) {
      throw new Error('the request was not counted');
    }
    const count = Number(row.count);
    if (count === limit.max + 1) {
      await writeSecurity({
        event: 'limit_hit',
        details: { limit: limit.name, max: limit.max, windowSeconds: seconds },
      });
    }
    return {
      count,
      refused: count > limit.max,
      retryAfterSeconds: Math.max(1, Number(row.retry_after)),
    };
  });
}
   ```
4. Create `apps/api/src/shared/address-limit.ts` with exactly this content:
   ```ts
import type { RequestHandler } from 'express';
import { AppError } from './errors.js';
import { hitLimit } from '../modules/core/repository/rate-counters.js';

// The per-address limit (TASK-0023, TECH-DESIGN section 2 step 2, NFR-020): on the sign-in doors, the two-step doors
// and, later, the public pages, every request from one address is counted, refused ones included, before anyone is
// known; past the limit the door answers 429 until the window ends. The values come from the environment file's
// ADDRESS_LIMITS, read at each request, as "<name>:<count>/<unit>" parts separated by commas, one for each name.
const NAMES = ['signin', 'public'] as const;
export type AddressLimitName = (typeof NAMES)[number];
const UNITS = new Map([
  ['second', 1],
  ['minute', 60],
  ['hour', 3600],
]);
const DEFAULT_LIMITS = 'signin:20/minute,public:60/minute';
const PART = /^([a-z]+):([1-9][0-9]{0,5})\/([a-z]+)$/;

export function addressLimits(
  value: string | undefined = process.env.ADDRESS_LIMITS,
): Record<AddressLimitName, { max: number; windowSeconds: number }> {
  const limits: Partial<Record<AddressLimitName, { max: number; windowSeconds: number }>> = {};
  for (const raw of (value?.trim() || DEFAULT_LIMITS).split(',')) {
    const part = raw.trim();
    const match = PART.exec(part);
    const name = match?.[1] as AddressLimitName | undefined;
    const windowSeconds = match ? UNITS.get(match[3]!) : undefined;
    if (!match || !name || !NAMES.includes(name) || windowSeconds === undefined) {
      throw new Error(`ADDRESS_LIMITS has a part that is not <name>:<count>/<unit>: ${part}`);
    }
    if (limits[name]) {
      throw new Error(`ADDRESS_LIMITS gives ${name} twice`);
    }
    limits[name] = { max: Number(match[2]), windowSeconds };
  }
  for (const name of NAMES) {
    if (!limits[name]) {
      throw new Error(`ADDRESS_LIMITS has no limit for ${name}`);
    }
  }
  return limits as Record<AddressLimitName, { max: number; windowSeconds: number }>;
}

// The address that is counted: the one the proxy saw (Express trusts exactly one proxy), without an IPv4 address's
// IPv6 prefix.
function addressOf(ip: string | undefined): string {
  return (ip ?? '').replace(/^::ffff:/, '').slice(0, 64) || 'unknown';
}

export function addressLimit(name: AddressLimitName): RequestHandler {
  return async (req, res, next) => {
    const { max, windowSeconds } = addressLimits()[name];
    const counted = await hitLimit({ name, key: `ip:${addressOf(req.ip)}`, max, windowSeconds });
    if (counted.refused) {
      res.setHeader('Retry-After', String(counted.retryAfterSeconds));
      throw new AppError(
        429,
        'too_many_requests',
        `Too many tries. Please try again in ${counted.retryAfterSeconds} seconds.`,
      );
    }
    next();
  };
}
   ```
